cmake_minimum_required(VERSION 3.20)
project(pacetun_tls VERSION 6.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PACETUN_PORTABLE_CORE_ONLY "Build shared core without a tunnel executable" OFF)
find_package(OpenSSL REQUIRED)
if(PACETUN_PORTABLE_CORE_ONLY)
  add_library(pacetun-portable-core STATIC src/config.cpp src/record.cpp)
  target_include_directories(pacetun-portable-core PUBLIC include)
  target_link_libraries(pacetun-portable-core PUBLIC OpenSSL::Crypto)
  if(MSVC)
    target_compile_options(pacetun-portable-core PRIVATE /W4)
  else()
    target_compile_options(pacetun-portable-core PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  include(CTest)
  if(BUILD_TESTING)
    add_executable(pacetun-portable-core-selftest tests/portable_core_test.cpp)
    target_link_libraries(pacetun-portable-core-selftest PRIVATE pacetun-portable-core)
    add_test(NAME portable_config_crypto_telemetry COMMAND pacetun-portable-core-selftest "${CMAKE_SOURCE_DIR}/config/iran.conf")
    add_executable(pacetun-portable-logging-selftest tests/logging_test.cpp)
    target_include_directories(pacetun-portable-logging-selftest PRIVATE include)
    add_test(NAME portable_human_logging COMMAND pacetun-portable-logging-selftest)
  endif()
  return()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "PaceTun TUN runtime is Linux-only. Use -DPACETUN_PORTABLE_CORE_ONLY=ON to test the portable core, NOT to build a tunnel.")
endif()
set(PACETUN_CORE src/main.cpp src/config.cpp src/platform/linux/tun_linux.cpp src/record.cpp src/runtime.cpp src/websocket.cpp src/carrier.cpp src/admission.cpp)
add_executable(pacetun-tls ${PACETUN_CORE})
target_include_directories(pacetun-tls PRIVATE include)
target_link_libraries(pacetun-tls PRIVATE OpenSSL::SSL OpenSSL::Crypto pthread)
target_compile_options(pacetun-tls PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS pacetun-tls RUNTIME DESTINATION bin)

  get_property(test_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)

      target_compile_options(${t} PRIVATE -UNDEBUG)
    endif()
  endforeach()
